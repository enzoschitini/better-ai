    #Primi passi con Agent UI

    Per clonare Agent UI, esegui il seguente comando nel tuo terminale:

    npx create-agent-ui@latest

    Digita y per creare un nuovo progetto e installare le dipendenze, poi avvia agent-ui usando:

    C:\Users\schit>cd agent-ui
    C:\Users\schit\agent-ui>npm run dev

    Apri http://localhost:3000 per vedere Agent UI, ma ricorda di collegarti ai tuoi agenti locali.

    Local: http://localhost:3000
    Network: http://10.5.0.2:3000

    #Configurazione minima

    import os
from dotenv import load_dotenv

#os.environ["AGNO_TELEMETRY"] = "false"
#AGNO_TELEMETRY = os.getenv("AGNO_TELEMETRY")

from agno.agent import Agent
from agno.models.openai import OpenAIChat
from agno.os import AgentOS

load_dotenv()

assistant = Agent(
    name="Assistant",
    model=OpenAIChat(id="gpt-4.1-mini"),
    instructions=["You are a helpful AI assistant."],
    markdown=True,
)

agent_os = AgentOS(
    id="my-first-os",
    description="My first AgentOS",
    agents=[assistant],
)

app = agent_os.get_app()

if __name__ == "__main__":
    # Default port is 7777, change with port=...
    agent_os.serve(app="minimal_setup:app", reload=True)
